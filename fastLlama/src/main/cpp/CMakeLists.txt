cmake_minimum_required(VERSION 3.22.1)
project(local_llama_fast C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod" CACHE STRING "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../../../build/llama-source ${CMAKE_CURRENT_BINARY_DIR}/llama)
add_library(local_llama_fast SHARED fast.cpp)
# Export the pinned C API from a self-contained shared object for dlsym.
target_link_libraries(local_llama_fast PRIVATE -Wl,--whole-archive llama ggml ggml-cpu ggml-base -Wl,--no-whole-archive log dl)
target_link_options(local_llama_fast PRIVATE -Wl,-z,max-page-size=16384 -Wl,-Bsymbolic)
